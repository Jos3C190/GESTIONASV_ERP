describe('AUDIT block 3 actual payloads', () => {
  it('renders cost_delta from a late retaceo', () => {
    const data = {...valuedData, entries: [{...valuedData.entries[0]!, kind: 'cost_adjustment', quantity_base: '0', inventory_delta: '8', cost_breakdown: [{event_id: 'out-1', kind: 'issue', cost_delta: '12', source_reference: 'SALE-1'}]}]};
    const {container} = render(ValuedKardexList, {props: {data, loading: false, onPageChange: vi.fn()}});
    expect(within(findMobileList(container)).queryByText(/Origen SALE-1.*USD 12/)).not.toBeNull();
  });
  it('renders supplier credit, outbound cost and difference separately', () => {
    const data = {...valuedData, entries: [{...valuedData.entries[0]!, kind: 'supplier_credit', quantity_base: '0', source_value: '100', inventory_delta: '0', cost_breakdown: [{event_id: 'return-1', kind: 'supplier_credit', credit_net: '100', inventory_cost: '75', cost_difference: '25', source_reference: 'RETURN-1'}]}]};
    const {container} = render(ValuedKardexList, {props: {data, loading: false, onPageChange: vi.fn()}});
    const list = within(findMobileList(container));
    expect(list.queryByText(/USD 100/)).not.toBeNull();
    expect(list.queryByText(/USD 75/)).not.toBeNull();
    expect(list.queryByText(/USD 25/)).not.toBeNull();
  });
});
